١٨٨٤ - (خ م) عن أنس، قال: كـان النبي ﷺ إذا غـزا بـنـا قـومـاً، لم يـكـن يـغـزو بـنـا حتى يـصـبـح وينـظر، فإن سـمع أذانـاً كـفَّ عـنهـم، وإن لـم يـسـمـع أذانـاً أغـار عـليـهـم. فخـرجـنـا إلى خـيـبـر، فـانـتـهـيـنـا إلـيـهـم ليـلاً، فـلـمـا أصـبـح ولـم يـسـمـع أذانـاً ركـب، وراكـبـتُ خـلـف أبي طـلـحـة، وإن قـدمـي لـتـمـس قـدم النبي ﷺ فخـرجـوا إلـيـنـا بـمـكـاتـلهـم ومـسـاحيـهـم، فـلـمـا رأوا النبي ﷺ قـالـوا: مـحـمـد والـلـه، مـحـمـد والـخـمـيـس، - قـال: والـخـمـيـس: الجـیـش - فـلـمـا رآهـم رسول الله ﷺ قـال: الله أكـبر، الله أكـبـر، خـَرِبـت خـيـبـر، إنـا إذا نـزلـنـا بـسـاحة قـوم فـسـاء صـبـاح الـمـنـذَرین، فـظـهـر رسول الله ﷺ عـليـهـم، فـقـتـل الـمـقـاتلـة، وسـبى الذراري.
(وفي رواية): أنَّ رسول الله ﷺ غـزا خـيـبـر، قال: فـصـلـَّيـنا عـنـدها صـلاة الـغـدَاةِ بـغـَلـَس، فـركـب النبيُّ ﷺ وركـب أبو طـلـحـة، وأنا رَدِيفُ أبي طـلـحـة، فـأجـرى نـبيُّ الله ﷺ في زُقـَاق خـيـبـر - وإنَّ ركـبتي لـتـمـسُّ فَخـِذَ نـبـيِّ الله ﷺ وانـْحـَسـَرَ الإزارُ (وفي رواية): ثم حـَسـَرَ رسولُ الله ﷺ الإزارَ عـن فخـذه، حتى إني أنـظـر إلى بيـاض فَخـِذ نـبـيِّ الله ﷺ فـلـمـا دخـل الـقـرية قـال: الله أكـبـر، خـَرِبـت خـيـبـر، إنـا إذا نـزلـنـا بـسـاحة قـوم، فـسـاء صـبـاحُ الـمـنـذَرین - قـالـها ثلاث مـرات - قال: وقـد خـرج الـقـوم إلى أعـمـالهـم، فـقـالـوا: مـحـمـد والـخـمـيـس، قال: وأصـَبـْنـاها عـَنـْوَة.
(عـَنـْوَة: قـهـراً بـغيـر صـلـح، كـما يـقـال: أخـذهـا بـالسيـف، قـوله: فـأجـرى، أي أجـرى مـركـوبـه. والـزقـاق: السكـة أو الـطريـق. واستـدل مـالـك بـهذا ورواية عـن أحـمد عـلى أن الـفخـذ لـيـست عـورة، وعـنـد أكـثـر الـعـلـمـاء أنـها عـورة ویـحمـلون الحديث عـلى أن الإزار انحسـر للـزحـمة وإجـراء الـمركوب ووقـع نـظر أنـس إليـه فـجـأة وكـذلك مـسُّ رك_بته الـفخـذ مـن غـيـر اخـتيارهـما، انتـهـى كـلام الـنووي مـلخصـاً، وقال الـخطـابي: فيـه أن قـتـال الـكفـار مـن غـي_ر إحـداث الـدعـوة جـائز، وأن الأذان شـعـار الإسـلام وأنـه لا يـجوز تـركـه ولـو أن أهـل بـلـد اجـتـمعوا عـلى ت_ركه كـان للسلطـان قـتـالـهم عـلـيـه).
