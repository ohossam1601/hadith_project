١٨٦٥ - (خ م) عن عائشة، قالت: أُصـيـب سـعدٌ يوم الخندق، رماه رجل من قريش، يقال له: حيـَّان بن العـَرِقة، رماه في الأكـْحـَل، فضرب عليه رسول الله ﷺ خـيـمـة في المسجد، لـيـعـوده من قريب، فلما رجع رسول الله ﷺ من الخـنـدق، وضع السـلاح واغتـسل، فأتـاه جبريلُ وهو يـنـْفـُض رأسه من الغبار، فقال: قد وضـعـتَ الـسـلاح؟ والله مـا وضعـتـُه (وفي رواية: والله مـا وضـعناه) اخرج إليهم، فقال النبي ﷺ: فأين؟ فأشار إلى بني قريظة، فأتاهم رسولُ الله ﷺ فنـزلوا على حـُكـْمِـه، فـردَّ الحـُكـْم إلى سـعد، قال: فإني أحـكـُم فيهم: أن تـُقتـل المقاتـلـةُ، وأن تـُسـبـى النساءُ والـذُّرِّيـّة، وأن تـُقـسـَّم أموالهم. قال هشام: فأخبرني أبي عن عائشة أن سعداً قال: اللـهم إنك تعلم أنه ليس أحد أحـبَّ إليَّ أن أجـاهـدَهم فيـك من قوم كـذَّبوا رسولك وأخـرجـوه، اللـهم فإني أظن أنك قد وضعتَ الحرب بيننا وبينهم، فإن كان بقي من حرب قريش شـيـء فـأبـقـني لهم حتى أجـاهـدَهم فيك، وإن كنتَ وضعتَ الـحـرب فافـجـُرْها واجعل مـوتي فيها، فانفجرت من لـَبـَّته (وفي رواية: من لـيـلـتـه) فلم يـَرُعـْهم - وفي المسجد خيمة من بني غـفـار - إلا الـدَّمُ يـسيـل إليههم، فقالوا: يا أهل الخـيـمـة، ما هذا الذي يأتينا من قـِبـلـِكم؟ فإذا سـعـدٌ يـَغـْذُو جـُرحـُه دماً، فـمات منها.
{(قال ابن حجر: حيَّان بكسر المهملة وتشديد الموحدة، والعـَرِقـة بفتح المهملة وكسـر الراء ثـم قاف، واسـمـه جـبـَّان بن قـَيْس، والعـَرِقـة أُمـُّهُ: العـَرِقة بنت سـعيـد. الأكـْحـَل: عـِرْق في وسط اليد. غـذا الجـرح - بالذال المعجمة - يـغـْذُو غـَذْواً: إذا دام سـيـلانـه}.
