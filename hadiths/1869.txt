١٨٦٩ - (خ) عن عـُروة بن الـزبـيـر، عـن الـمـِسـْوَر بن مـَخـْرَمـة ومـَـروان - يُـصـدِّق كلُّ واحـد مـنـهما حـديث صـاحبـه - قالا: خـرج النبيُّ ﷺ زَمـَنَ الحـُدَيْـبـية، حتى إذا كانوا ببعض الطريق قال النبيُّ ﷺ: إنَّ خـالـدَ بنَ الولـيـد بـالغـُـمـَيـم في خـيـل لقـريش طـلـيعـة، فـخـذوا ذاتَ اليـمـين، فوالله مـا شـَعـر بهم خالد، حتى إذا هُمْ بـقـَتـَرَة الجـيـش، فـانـطـلـق يـركـُضُ نـذيراً لقريش، وسـار النبيُّ ﷺ حتى إذا كان بالثـَّنـِيـَّة التي یـهـْبـط عليهم منها بـَرَكَـتْ به راحـلـتُه، فقال الناس: حـَلْ حـَلْ، فـألـَحـَّـتْ، فقالوا: خـلأتِ الـقـصـواءُ، خـلأت القـصواء، فقال النبيُّ ﷺ: ما خـلأت القصواء، وما ذاك لها بـخـُـلق، ولكن حـبـَـسها حـابـسُ الفـيـل، ثم قال: والذي نفسي بيده لا يسألوني خـُطـَّة يعـظـمون فيها حـُرُمـاتِ الله إلا أعطيتـهم إياها، ثم زجرها فـوثـبـتْ، فـعـدَلَ عنهم حتى نزل بأقصى الحـُدَيْـبية على ثـَمـَد قليل الماء، يـتـبـرَّضـُه الناسُ تـبـرُّضـاً، فلم يلبث الناسُ حتى نزحـوه، وشُـكـِيَ إلى رسول الله ﷺ العـطـشُ، فـانـتـزع سـهـماً من كِـنـانتـِه، ثم أمرهم أن يجعلوه فيه، فوالله مـا زال يـجيش لهم بالـرِّيِّ حتى صـدَروا عنه، فبينا هم كذلك إذ جـاء بـُدَيْـل بنُ وَرْقـاءَ الخزاعيُّ في نفر من قومه من خـُزاعـةَ - وكانوا عـَيْـبـَةَ نـُصـْح رسول الله ﷺ من أهل تِـهـامـةَ - فقال: إني تركتُ كـَعـْبَ بنَ لـُؤي و عامر بن لـؤي نزلوا أعـدادَ مياه الحـديبيـة، معهم العـُـوذُ المـَطافـيـل، وهم مـقاتـلوكَ، وصـادُّوك عن البيت، فقال رسولُ الله ﷺ: إنا لم نـجيئ لـقـتال أحـدٍ ولكنـَّا جئنا مـُعـتـمـرین، وإنَّ قريشاً قد نـهـكـتـْهم الحربُ وأضرَّتْ بهم، فإن شـاؤوا مـادَدْتـُهـم مـُدَّةً، ويـُخـلـوا بيـني وبين الناس، فإن أظهرُ عليهم، فإن شـاؤوا أن يدخلوا فيـما دخـل فيـه الناس فـعلووا، وإن هم أبوا، فـوالذي نفسـي بيده لأقـاتـلـنـَّهم على أمري هذا، حتى تـنـْفـَرِدَ سـالِـف-َتي، ولـيـنـفـِذنَّ الله أمره، فقال بـُدَيـل: سـأبـلـغـهم ما تقول، فـانطلق حتى أتى قريشاً، فقال: إنا قد جئناكم من هذا الرجل، وقد سمعناه يقول قـولاً، فإن شئتم أن نعـرضـهُ عليكم فعـلـنا، فقال سفـهاؤهم: لا حاجة لنا أن تـُخبرنا عنه بشيء، وقال ذوو الرأي منهم: هاتِ ما سمعتـَه يقول، قال: سمعتـَه يقول كذا وكذا - فـحـدَّثـهم بما قال النبيُّ ﷺ فقام عروة بن مسعود، فقال: أي قوم، ألـسـتم بالوالد؟ قالوا: بلى، قال: أوَلـَسـتُ بالـوَلـد؟ قالوا: بلى، قال: فهل تـتـهموني؟ قالوا: لا، قال: ألسـتم تعلمون أني اسـْتـنـفـرتُ أهلَ عـُكـاظ، فلما بـَلـَّحـوا عليَّ جئـتـكم بأهلي وولدي ومن أطاعني؟ قالوا: بلى، قال: فإنَّ هذا قد عرض عليكم خـُطـَّة رُشـْد، اقـبـلوها، ودعوني آتـِهِ، قالوا: ائـتـِهِ، فـأتـاه، فـجعل يـكـلـّم النبيَّ ﷺ فـقال له النبيُّ ﷺ نـحـواً من قولـه لـبـُديـل، فقال عـروةُ عـند ذلك: أيْ مـحـمدُ، أرأيـتَ إن اسـْتـَـأصـَلـتَ أمـر قـومـك، هل سـمعـتَ بـأحد من العرب اجـتـاح أصـلَـه قـبـلـَك؟ وإن تـكن الأخـرى، فـإني واللـه لأرى وجـوهاً، وإني لأرى أوشـابـاً من الناس، خـَلـيـقاً أن يـفـرُّوا ويـدَعـُوك، فقال له أبو بكر: امـْصـص بـبـَظـْرِ اللات، أنـحـنُ نـَفِـرُّ عنـه و نـدَعـُه؟ فقال: مـَن ذا؟ قالوا: أبو بكر، فقال: أما والذي نفسي بيده، لولا يـدٌ كانت لك عندي لم أجـْزِك بها لأجـَبـتـك، وجـعل يـكـلـّم النبيَّ ﷺ فـكـلمـا كـلـَّمـه أخذ بلحيـتـه، والمغيـرةُ بنُ شـعـبة قائم على رأس النبي ﷺ ومعه السيف، وعليه المـِغـْفـَرُ، فكلمـا أهـْوَى عـروةُ بيـده إلى لحيـة رسول الله ﷺ ضرب يـدَه بـنـَعـْلِ السيف، وقال: أخر يـَدَك عن لحيـة رسول الله ﷺ فـرفع عروةُ رأسـَه، فقال: من هذا؟ قالوا: المغيـرة بن شعبة، فقال: أي غـُدَرُ ألـستُ أسـعى في غـَدْرَتـِك؟ - وكان المغيـرةُ صـحب قـوماً في الجـاهلية، فـقتلهم وأخذ أمـوالهم، ثم جـاء فـأسـلـم، فقال النبي ﷺ: أمـا الإسـلامُ فـأقـبـَلُ، وأما المـالُ فـلـستُ منه في شيء - ثم إن عـروة جـعل يـَرْمـُقُ أصحـاب النبي ﷺ بعيـنيـه، قال: فـواللـه ما تـنخَّـم رسولُ الله ﷺ نـُخامة إلا وقـعت في كفّ رجل مـنهم فـدلـَّك بها وجـهه وجـلـدَه، وإذا أمرهم ابـتـدروا أمـره وإذا تـوضـأ كـادوا يـقتـتلون على وضوئه، وإذا تـكلم خـفـضوا أصواتـهم عـنده، وما يـُحِـدُّون إليه النظر تـعـظيماً له، فـرجع عـروة إلى أصـحابه فقال: أي قوم، والله لقـد وفـَدتُ على الـملـوك، ووَفـَدتُ على كـسـرى وقـيـصـر والنجـاشـي، والله إن رأيـتُ مـَلـِكاً قـطُّ يـعـظـّمهُ أصحـابـُه ما يـعـظـّم أصحابُ مـحـمدٍ مـحمداً، والله إن تـنـخـم نـخامة إلا وقعت في كف رجل مـنهم فـدَلـَّك بها وجهه وجلده، وإذا أمرهم ابتدورا أمره، وإذا توضأ كادوا يقتتلون على وضوئه، وإذا تـكـلـّم خـفـضـوا أصـواتـهم عـنده، وما يـُحِـدُّون إليه النظر تـعـظـيماً له، وإنه قد عـرَضَ عليكم خـُطـَّة رُشـْد فـاقـبـلوها، فقال رجل من بني كِـنانـة: دعوني آتـِهِ، فقالوا: ائـتـه، فلما أشـرف على النبي ﷺ وأصحابه، قال رسولُ الله ﷺ: هذا فلان، وهو من قوم يـُعـظـمون الـبـُدْنَ، فـابـعـثـوها له، فـبـُعـثـت له واستـقـبـله الناسُ يُـلـبـون، فلما رأى ذلك قال: سبحان الله! ما ينبغي لـهؤلاء أن يُـصـدُّوا عن البيت، فلما رجع إلى أصحابه قال: رأيتُ البـُدْنَ قد قـُلـِّدتْ وأُشـْعِـرَتْ، فما أرى أن يـُصـدُّوا عن البيت، فقام رجل منهم يقال له: مـِكـْرَزُ بنُ حفص، فقال: دعوني آتـِه، فقالوا: ائته، فلما أشرف عليهم، قال النبيُّ ﷺ: هذا مـِكـْرَزُ بنُ حفص، وهو رجل فاجر، فجعل يـكـلـّم النبيَّ ﷺ فبينا هو يـكـلـّمه، إذ جاء سـهيـل بن عمرو - قال معمر: فـأخـبرني أيوب عن عكرمة: أنه لما جاء سهيل، قال النبيُّ ﷺ: قد سـهـُل لكم من أمركم. قال معمر: قال الزهريُّ في حديثه: فجاء سهيل بن عمرو، فقال: هاتِ اكتـُبْ بينـنا وبينك كتاباً، فدعا النبيُّ ﷺ الكاتب، فقال النبيُّ ﷺ: اكتب: بسم الله الرحمـن الرحيم، فقال سهيل: أما الرحمـن، فوالله ما أدري ما هو؟ ولكن اكتب: باسمك اللـهم، كما كنتَ تكتب، فقال المسلمون: والله لا نـكتـُبـها إلا بسم الله الرحمن الرحيم، فقال النبيُّ ﷺ: اكتب: باسمك الـلـهم، ثم قال: هذا ما قاضى عـليه مـحـمد رسولُ الله، فقال سهيل: والله لو كـنـَّا نعلم أنـَّك رسولُ الله ما صـدَدْنـاك عن البيت، ولا قـاتـلـنـاك، ولكن اكتب: محمدُ بنُ عبدالله، قال النبيُّ ﷺ: والله إني لرسولُ الله، وإن كـذَّبـتـموني، اكتب: محمد بن عبدالله، قال الزهريُّ: وذلك لقوله: لا يسألوني خـُطـَّة يُـعـظـمون فيها حرمات الله إلا أعطيتـهم إياها، فقال له النبيُّ ﷺ: على أن تـُخـلـّوا بيـنـنا وبيـن البيت فنطوف به، فقال سهيل: والله لا تـَتـحـدَّثُ العربُ أنـَّا أُخـِذْنـَا ضـُغـْطـَة، ولكن ذلك من العـام المـقـبل، فـكتـب، فـقال سهيل: وعلى أنـَّه لا يـأتـيـك مـِـنـَّا رَجـُل - وإن كان على ديـنـك - إلا رَدَدْتـَه إليـنا، قال المسلمون: سبحان الله! كيف يُـرَدُّ إلى المشركين، وقد جـاء مسلماً؟ فبيـنا هم كذلك، إذ جـاء أبو جـَنْـدَلِ ابنُ سـهـيـل بن عمرو يـَـرْسـُف في قيـوده، وقد خرج من أسـفـل مكة، حتى رمى بنفسه بين أظـهر المسلمين، فقال سـهـيـل: هذا يا محمد أول ما أُقـاضـيـك عليه: أن تـرُدَّه إلـيَّ، فقال النبيُّ ﷺ: إنا لم نـَقـْضِ الـكتـاب بـعـدُ، قال: فـواللـه إذاً لا أُصـالـحـُك على شـيـء أبداً، فقال النبيُّ ﷺ: فـأجـِزْه لي، قال: مـا أنا بمُـجيـِزِه لك، قال: بلى فـافـعـل، قال: ما أنا بـفـاعل، قال مـِكـْرَزُ بنُ حـفـص: بـلى، قد أجـزناهُ لك، قال أبو جـندل: أي معـشر المسلمين، أُرَدُّ إلى المشركين وقد جئـتُ مسلماً؟ ألا تـرَوْنَ ما قد لقـيـتُ؟ وكان قد عـُذِّب عـذاباً شـديـداً في اللـه، قال: فـقال عمـرُ بنُ الخطـاب: فـأتيـتُ نـبيَّ الله ﷺ فقلتُ: ألـسـتَ نـبيَّ الله حقاً؟ قال: بلى، قلتُ: ألسنا على الحـقّ وعدُوُّنا على الباطل؟ قال: بلى، قلتُ: فـلِـمَ نـُعـْطي الـذُّنـِيـَّـةَ في ديننا إذاً؟ قال: إني رسولُ الله، ولـستُ أعـصـيـه، وهو نـاصري، قلتُ: أو ليس كنتَ تـحـدِّثـنـا أنـا سـنـأتي البيت و نطوف به؟ قال: بلى، قال: فـأخبرتـُك أنـَّا نـأتيه العام؟ قلت: لا، قال: فـإنـك آتـيه ومـُطـَّوِّفٌ به، قال: فـأتيـتُ أبا بكر، فـقلتُ: يا أبا بكر، أليس هذا نـبيُّ اللـه حقاً؟ قال: بلى، قلتُ: ألسنا على الحـق، وعـدُّونـا على الباطل؟ قال: بلى، قلتُ: فـلم نـُعـطي الـدَّنـيةَ في ديـنـنا إذاً؟ قال: أيـُّها الرجل، إنـَّه رسولُ الله ﷺ وليـس يعـصـي ربـَّه، وهو نـاصـرُه، فـاستـَمـسك بـِغـَرْزِه، فـوالـلـه إنـَّه عـلى الحـق، قلتُ: أو لـيـس كان يـحـدِّثـنـا: أَنـَّا سـنـأتـي البـيـت ونـطوفُ به؟ قال: بلى، أفـأخـبـرك أنـك تـأتـيه الـعـام؟ قلت: لا، قال: فـإنـك آتيه ومـُطـَّوِّف بـه؟ قال عـمر: فـَعـَمـِلـتُ لذلك أعـمـالاً، قال: فلما فرغ من قضية الكتاب، قال رسول الله ﷺ لأصحابه: قوموا فانـحـروا، ثم احـلـقـوا، قال: فـواللـه ما قام مـنهم رجل حتى قال : ذلك ثلاث مرات، فلما لم يـقـمْ مـنهم أحد دخل على أمِّ سـَلـَمـةَ، فـذكر لـها مـا لـقـيَ من الناس، قالتْ أمُّ سـلمـةَ: يا نـبيَّ الله، أتـُحـِبُّ ذلك؟ اخرج، ولا تـكـلـّم أحـداً منـهـم كـلمة حتى تـنحـر بـُدْنـَك، وتـدعو حـالـِقـَك فـيـحـلقـك، فـخرج فـلم يـكـلـّم أحـداً مـنهـم حتى فـعل ذلك، نـحـر بـُدنـَه، ودعـا حـالـقـه فحلـقه، فـlـما رأوا ذلك، قاموا فـنـحـروا، وجـعـل بـعـضـهـم يـحـلـِق بـعـضـاً، حتى كـاد بـعـضـهـم يـقـتـل بـعـضـاً غـَمـَّاً، ثم جـاءه نـسـوة مؤمنات، فـأنـزل اللـه ﷻ: ﴿يَا أَيُّهَا الَّذِينَ آمَنُوا إِذَا جَاءَكُمُ الْمُؤْمِنَاتُ مُهَاجِرَاتٍ فَامْتَحِنُوهُنَّ﴾ حـتـَّى بـلـغ ﴿بِعِصَمِ الْكَوَافِرِ﴾ فـطـلـَّق عمرُ يومئذ امرأتـين كـانـتا لـه في الشـِّرك، فـتـزوج إحـداهـما مـعـاويـةُ بنُ أبي سفيان، والأخـرى صـفـوانُ بن أُ مـَيـَّة، ثم رجع النبيُّ ﷺ إلى المـديـنة، فـجاءه أبو بـصـيـر - رجل من قريش - وهو مـسلمٌ، فـأرسـلوا في طلـبـه رَجـُلـَيـن، فـقالوا: الـعـَهـْد الذي جـعـلـتَ لنـا، فـدفـعـه إلى الرجليـن، فـخرجـا بـه، حـتى بـلـغا ذا الحـُـلـيـفـةِ، فـنـزلـوا يـأكلون من تـَمـر لـهم، فـقال أبو بـصـيـر لأحـد الرجليـن: والله إني لأرى سـيـفـك هـذا جـيـداً، فـاستـلـَّه الآخـرُ، فـقال: أجـل، والله إنـه لـَجيـد، لقد جـرَّبـتُ بـه، ثم جـرَّبـتُ، فـقال أبو بـصـيـر: أرني أنـظر إليه، فـأمكـنـه منه، فضربه حتى بـَرَدَ، وفـرَّ الآخرُ حتى أتى المدينة، فـدخل المـسـجد يـعـدُو، فقال النبيُّ ﷺ حيـن رآه: لقد رأى هـذا ذُعـْراً، فلما انتهى إلى النبي ﷺ قال: قـُتـِل والله صـاحـبـي، وإني لـَمـقـتـول، فـجاء أبو بـصـيـر، فقال: يا نـبيَّ الله، قد والله أوفـى الـلـه ذِمـَّتـَك، قد رددتـني إليهـم، ثم أنـجـاني الله منهم، فقال النبيُّ ﷺ: ويـلُ أمـّه، مـِسـْعـَر حـَرْب، لو كان له أحـد، فلما سمع ذلك عَـرَف أنـه سـيـرُدُّه إليههم، فـخرج حتى أتى سـيـف البحر، قال: ويـنفلتُ مـنهم أبو جـندل بن سهيل فـلحق بـأبي بـصـير، فكان لا يخرج مـن قريش رجل قد أسلم إلا لحـق بـأبي بـصيـر، حتى اجتمعت مـنهم عصابة، فوالله ما يسمعون بـعـير خـرجـت لقريش إلى الشـام، إلا اعترضوا لها، فقتلوهم وأخذوا أموالهم، فأرسلت قريش إلى النبي ﷺ تـُناشدُه الله والـرَّحـِمَ لمـا أرسـل إليههم، فمـَن أتـاه مـنهم فـهـو آمـن، فأرسل النبيُّ ﷺ إليهم، فـأنـزل الله ﷻ: ﴿وَهُوَ الَّذِي كَفَّ أَيْدِيَهُمْ عَنكُمْ وَأَيْدِيَكُمْ عَنْهُم بِبَطْنِ مَكَّةَ مِن بَعْدِ أَنْ أَظْفَرَكُمْ عَلَيْهِمْ﴾ حـتـَّى بـلـغ ﴿الْحَمِيَّةَ حَمِيَّةَ الْجَاهِلِيَّةِ﴾ وكـانـت حـمـيـّتـُهم: أنـهم لـم يـُقـِرُّوا أنـه نـبيُّ الله، ولـم يـُقـِرُّوا ببسم الله الرحمن الرحيم، وحالوا بينهم وبين البيت.

(وفي رواية): أن عـروة سمع مـَرْوان والـمـِسـْوَر يـُخـْبـِرَان عن أصـحاب رسول الله ﷺ قال: لما كاتب سـُهـَيـلُ بنُ عمرو يومئذ، كان فـيما اشترط سـهـيـل عـلى النبي ﷺ أنـه لا يـأتـيـك مـِـنـَّا أحـد وإن كان على ديـنـك إلا رددتـَه إلـيـنا، وخـلـَّيتَ بـيـنـنا وبيـنه، فـكـره المؤمنون ذلك، وامـتـعضـوا مـنه، وأبـى سـهـيـل إلا ذلك، فـكاتبـه النبي ﷺ عـلى ذلك، فـردَّ يـومئـذ أبا جـنـدل إلى أبيـه سـهـيل بن عـمـرو، ولـم يـأتـِهِ أحـد من الرجـال إلا رَدَّهُ في تلـك الـمـُدَّةِ وإن كان مسـلمـاً، وجـاء المؤمنـاتُ مـهاجـرات، وكانت أمُّ كلثوم بـنـتُ عُـقـبة بن أبي مـُعـيـط مـمن خرج إلى رسول الله ﷺ يومـئذ وهي عاتـِق، فـجـاء أهلـها يسـألون النبي ﷺ أن يـرْجـِعـَها إليههم، فـلـم يـرْجـِعـها إلـيـهـم، حتى أنـزل الله فيـهنَّ: ﴿إِذَا جَاءَكُمُ الْمُؤْمِنَاتُ مُهَاجِرَاتٍ فَامْتَحِنُوهُنَّ ۖ اللَّهُ أَعْلَمُ بِإِيمَانِهِنَّ ۖ فَإِنْ عَلِمْتُمُوهُنَّ مُؤْمِنَاتٍ فَلَا تَرْجِعُوهُنَّ إِلَى الْكُفَّارِ ۖ لَا هُنَّ حِلٌّ لَّهُمْ وَلَا هُمْ يَحِلُّونَ لَهُنَّ﴾.
