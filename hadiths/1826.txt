۱۸۲٦ - (م) عن زيد بن وهب الجُهني، أنه كان في الجيش الذين كانوا مع عليٍّ رضي الله عنه، الذين ساروا إلى الخـوارج. فقال عـلـي رضي الله عنه: أيـهـا الناس، إني سـمـعـتُ رسول الله ﷺ يقول : يـخـرج قـوم مـن أُمـتـي يـقـرؤون الـقـرآن، لـيـس قـراءتـكـم إلى قـراءتـهـم بـشـيء، ولا صـلاتـكـم إلى صـلاتـهـم بـشـيء، ولا صـيـامـكـم إلى صـيـامـهـم بـشـيء، يـقـرؤون الـقـرآن، يـحـسـبـون أنـه لـهـم وهـو علـيـهـم، لا تـجـاوز صـلاتـهـم تـراقـيـهـم، يـمـرقـون مـن الإسـلام كـمـا يـمـرق الـسـهـم مـن الـرمـيَّة، لـو يـعـلـم الـجـيـشُ الـذي يـصيـبـونـهـم مـا قـُضيَ لـهـم عـلى لسـان نـبـيـهـم ﷺ لاتـكـلـوا (وفـي لـفـظ: لـنـكـلوا) عـن الـعـمـل، وآيـةُ ذلك أنَّ فـيـهـم رجـلاً لـه عَـضـُدٌ، ولـيـس لـه ذراع، عـلى رأس عـَضـُدِه مـثـل حـَلـَمـَة الـثـدي، عـليـه شـعـرات بـيـضٌ، فـتـذهـبـون إلى مـعـاويـة وأهـل الـشـام وتـتـركون هـؤلاء يـخـلـفـونـكـم في ذراريـكـم وأمـوالـكـم، والله، إني لأرجـو أن يـكـونـوا هـؤلاء الـقـوم، فإنـهـم قـد سـفـكـوا الـدَّمَ الـحـرام، وأغـاروا في سـَرْح الـناس، فـسـيـروا عـلى اسـم الله. قـال سـلـمةُ بـن کـهـيـل: فـنـزلني زيـدُ بـن وهـب مـنـزلاً مـنـزلاً، حتى قـال : مـَرَرْنـا عـلى قـنـطـرة، فـلـمـا الـتـقـيـنـا وعـلـى الـخـوارج يـومـئـذ عـبـدالله بـن وهـب الـراسبـي، فـقـال لـهـم: ألقـوا الـرِّمـاح، وسـُـلـُّوا سـيـوفـكـم مـن جـفـونـها فـإنـي أخـاف أن يـُـنـاشدوکـم كـمـا نـاشـدوكـم يـوم حـروراء. فـرجـعـوا فـوحـشـوا بـرمـاحـهـم، وسـلـوا الـسـيـوف، وشـجـرهـم الـناس بـرمـاحـهـم، وقـُتـل بـعضـهـم عـلى بـعـض، ومـا أُصيـب مـن الـنـاس يـومـئـذ إلا رجـلان، فـقـال عـلـي رضي الله عنه : الـتـمـسـوا فـيـهـم الـمـُـخـدَجَ، فـالـتـمـسـوه فـلـم يـجـدوه. فـقـام عـلـي رضي الله عنه بـنفـسـه حـتى أتـى ناسـاً قـد قـُـتـل بـعـضـهـم عـلى بـعـض، قـال : أخـروهـم، فـوجـدوه مـمـا يـلي الأرض، فـكـبـَّر، ثـم قـال : صدق الله، وبـلـّغ رسولـه. فـقـام إلـيـه عـبـيـدَةُ السـلـمـانـي، فـقـال : يـا أمـيـر الـمـؤمـنـيـن، آللـه الـذي لا إلـه إلا هـو، لـسـمـعـت هذا الـحـديث مـن رسـول الله ﷺ؟ فـقـال: إي، واللـه الـذي لا إلـه إلا هـو، حـتـى اسـتـحـلـفـه ثـلاثـاً وهـو يـحـلـف لـه (وفـي روايـة عـن عـبـيـدَة السـلـمـانـي، عـن عـلـي رضي الله عنه) قـال: فـيـهـم رجـل مـُـخـدَج الـيـَد، لـولا أن تـبـطـروا لـحـدثـتـكـم بـمـا وعـد الله الـذيـن يـقـتـلـونـهـم عـلى لسـان مـحـمـد ﷺ قـال : فـقـلـتُ: أنـتَ سـمـعـتَ هـذا مـن مـحـمـد ﷺ قـال : إي، وربِّ الـكـعـبـة - قـالـها ثـلاثـاً -.
{قـولـه: فـنـزلنـي مـنـزلاً مـنـزلاً، أي: ذكـر لـي مـراحـلهـم بـالجيـش مـنـزلاً مـنـزلاً حتى بـلـغ الـقـنـطـرة الـتـي كان الـقـتـال عـنـدهـا. فـوحـشـوا بـرمـاحـهـم أي رمـوا بـهـا بـعـيـداً. الـمـُـخـدَج، بـضـم الـمـيـم وسـكـون الـخـاء الـمـعـجـمـة وفـتـح الـدال أي: ناقص الـيـد}.
