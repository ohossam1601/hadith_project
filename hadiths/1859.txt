١٨٥٩ - (خ) عن جعفر بن عمرو بن أمية الضـَّمـْري، قال: خرجتُ مع عُبيد الله بن عدِيّ بن الخيار، فلما قدِمـنا حـمـص، قال لي عبيد الله: هل لك في وَحـْشيّ نسأله عن قتل حمزة؟ قلت: نعم، فأن كان وَحشيّ يسكن حمص، فسألنا عنه؟ فقيل لنا: هو ذاك في ظلّ قصره، كأنه حـَمـيـْتٌ، فجئنا حتى وقفنا عليه يسيراً، فـسـلـَّمنا، فردَّ السلام، وعبيد الله مُـعـْتـَـجـِرٌ بعمامته، ما يرى وَحشيّ إلا عـَيْنيـه ورجليه، فقال عبيد الله: يا وَحشيُّ، أتـعرفني؟ فنظر إليه، ثم قال: لا والله، إلا أني أعلم أن عدِيَّ بنَ الخيار تزوج امرأة يُـقال لها: أمُّ قـِتـال بنتُ أبي العـيـص، فولدت له غـلاماً بمكة، فـكنـتُ أسـتـرضـعُ له، فـحمـلـتُ ذلك الغلام مع أمه، فـناولـتـُها إياه، فـلكأني نظرتُ إلى قـَدَم- مَيـْك، فـكشـف عبيدُ الله عن وجهه، ثم قال: ألا تخبرنا بقتل حمزة؟ قال: نعم، إن حمزة قـَتـَل طـُعـَيْـمـة بن عدي بن الخيار بـبـدر، فقال لي مولاي جبيـر بن مُـطـْعـم: إن قـتـلت حمزة بعـمّـي فـأنـت حـرّ، فلما خرج الناسُ عام عـَيـْنـَيـْن - وعـيـنـيـن جبل بـحيال أحـد، بـيـنه وبينـه وادٍ - خرجتُ مع الناس إلى الـقـتـال، فلما أن اصـطـفـّوا للـقتال خرج سـبـاعٌ فقال: هل من مُـبـارِز؟ فـخرج إليه حمزة بن عبد المطلب، فقال: يا سـبـاعُ، يا ابنَ أمِّ أنـْمـارٍ مـُقـَطـِّعةِ الـبـُظـور، أتـُحـادُّ الله ورسولـَه؟ ثم شـدَّ عليه، فكان كأمـسِ الذاهب، وكـمـنـتُ لحمزة تحت صخرة، فلما دنا مني رميتـُه بـِحـَرْبـَتـي، فأضعها في ثـُنـَّتـِهِ، حتى خرجتُ من بين وَرِكـَيْـه، فكان ذلك العهد به، فلما رجع الناس رجعتُ معهم، فـأقـمتُ بمكة حتى فـشا فيها الإسلام، ثم خرجتُ إلى الطائف، فـأرسـلوا إلى رسول الله ﷺ رُسـُلاً، وقيل لي: إنه لا يـَهـيـجُ الرسل، فـخرجتُ معهم، حتى قـَدِمـتُ على رسول الله ﷺ فلما رآني قال: أنت وحشيُّ؟ قلت: نعم، قال: أنت قتلت حمزة؟ قلتُ: قد كان من الأمر ما بلغك، قال: فهل تستطيع أن تـُغـَيـِّب وجهك عني؟ فخرجت، فلما قـُبـِض رسول الله ﷺ فخرج مسيلمةُ الكذاب قلتُ: لأخرجنَّ إلى مسيلمة لعلي أقتله، فـأكـافـئ به حمزة، فـخرجتُ مع الناس، فكان من أمره ما كان، فإذا رجل قائم في ثـُلـْمـَةِ جدار كأنه جمل أوْرَقُ، ثائـرُ الرأس، فرميتـُه بحربتي، فأضعها بين ثـَدْيـَيْه حتى خرجت من بين كتفيه، ووثب رجل من الأنصار، فضربه بالسيف على هامَتِـهِ، قال عبدالله بن الفضل: فأخبرني سليمان بن يسار: أنه سمع عبدالله بن عُـمـر يقول: فقالت جارية على ظهر بيـت: وَا أميـرَ المؤمنين، قتله العبد الأسودُ.
{الحـَمـيـْت: وعاء السـّمـن كالعـُكـَّة، وقيل هو الوعاء الـمـُمـتـَّن بالرُّبِّ. الاعـْتـِجـار: لـَفُّ العمامة على الرأس، ورد طرفها على الوجه. عام عينين: سنة أحد وعينين جبل بحيال أحد ونسبه إليه لأن قريشاً نزلوا عنده يوم أحد. سـبـاع: هو سباع بن عبدالعزى الخزاعي. مـُقـَطـِّعة الـبـُظـور: بـُظور النساء عند الختان. في ثـُنـَّتـِهِ: في عانته. لا يـَهيج الرسل: لا يزعجهم ولا ينفرهم. الـثـُّلـْمـة: الخلل في الجدار وغيره. أوْرَق، الـوُرْقـة في ألوان الإبل: كالسـُّمـْرة في الإنسان}.
