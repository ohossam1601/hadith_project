١٨٨٦ - (خ م) عن أبي حـُمـيـد الساعـدي، قال: غـزونـا مـع النبي ﷺ تـبـوك وأهـدى مـلـِكُ أيـْلـَةَ للنبي ﷺ بـغـلة بـيـضـاء، وكـسـاه بـُرْداً، وكـتـب لـه بـبـحـرهـم (هـذا لـفظ الـبخاري) (ولمسلـم) قال: خـرجـنـا مـع رسول الله غـزوة تـبـوك فـأتـيـنا وادي الـقـُرى، قال: وجـاء رسول ابـن الـعـُلـَمـاء، صـاحـب أيـْلـَةَ، إلى رسول الله ﷺ بـكـتـاب، وأهـدى لـه بـغـلة بـيـضـاء، فـكتـب إليـه رسول الله ﷺ بـبـحـرهـم، وأهـدى لـه بـُرْداً.
(سـبق الحديث بـطوله في كـتـاب ذكر رسول الله ﷺ. أيـْلـَة: تـسمى الـيـوم الـعـقـبة ویـسـمى جـزء مـنهـا أيـلات. قـوله: كـتـب لـه بـبـحـرهـم: أي: بـبلـدهـم أي: أنـه أقـرره عـلـيـهم بـمـا الـتـزموه مـن الـمـجزية، قـالـه ابـن حـجـر، وقـد تـقـدم أن الـعرب تـسمى الـفـُرى الـبـحـار).
