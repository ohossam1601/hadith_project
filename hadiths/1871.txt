١٨٧١ - (خ م) عن البـراء بن عازب، قال: اعـْتـَمـَر رسولُ الله ﷺ في ذي الـقـعـدة، فـأبـى أهلُ مـكـةَ أن یـَدَعـُوه يدخـلُ مـكـة، حتى قـاضـاهم على أن يـدخـل - يعـني من العام المـقـبل - يـُقـِيـّمُ فيها ثلاثة أيام، فلما كـتـبـوا الـكـتـاب، كـتـبـوا: هـذا ما قـاضـى عليه مـحمـد رسولُ الله، قالوا: لا نـُقـِرُّ بها، فلو نعـلـمُ أنـَّك رسولُ الله ما منعـناك، ولكن أنـتَ مـحـمدُ بنُ عبدالله، فقال: أنا رسول الله، وأنا مـحمدُ بنُ عبدالله، ثم قال لِـعـَلِيٍّ: امـْحـُه، فقال علي: مـا أنـا بالذي أمـْحـوه، فـمحـاه رسولُ الله ﷺ بيده، فـكـتـَب: هـذا ما قـاضـى عـليه مـحـمدُ بنُ عبدالله: لا یـَدْخـُلُ مـكة السـلاح إلا السـيـفَ في الـقـِراب، وأن لا يَـخـْرُجَ من أهلـها بـأحـد إن أراد أن يـَتـَّبـعـه، وان لا يـمـنع من أصحـابه أحـداً إن أراد أن يـُقـيـم بها، فلما دَخـلـها ومضـى الأجلُ أتـَوْا عـلـيـاً، فقالوا: قـل لصـاحبك: اخـرج عـنـَّا، فقـد مضـى الأجل، فـخـرج النبيُّ ﷺ (وفي رواية) قال: صـالـَحَ النبيُّ ﷺ المشركين يوم الحـديبيـة على ثلاثة أشياء: على أن مـَن أتـاه من المشركين رَدَّه إليهـم، ومـَن أتـاهـم من المسلميـن لم يـرُدُّهوه، وعلى أن يـدخـلـَهـا من قـابـل، ويـُقـيـم بها ثلاثة أيام، ولا يـدخـلـها إلا بـجـُـلـُبـَّان السـلاح - السيف والقوس ونحوه - فجاء أبو جـنـدل يـَـحـْجـُلُ في قـيـُوِدِه، فـردَّ إليهم.
{(تسمى تلـك العمرة: عـمـرة الـقـضاء؛ لأن رسول الله ﷺ قـاضـاهم عليها. يـحـجـل بضم الجيم وكسرها: يـقـفـز، قال الأزهري: والحـجلان مشـية الـمقـيد. الـقـِراب والجـُـلـُبـَّان: شبه الـجـِراب من الجـلد يضع الراكب فيـه سـيـفه بـجـفـنه، وسـوطـه وعـصـاه وأداتـه، وقولـه السـيف والقـوس ونحـوه، يـريـد ما يـحـتـاج في إظـهـاره إلى مـعـانـاة لا كـالـرماح لأنها ظـاهـرة يـمـكن تعـجـيـل الأذى بها، وإنـمـا اشتـرطـوا ذلـك ليـكون علامة لـلسـلـْم إذ كان دخـولـهم صـلـحـاً}.
