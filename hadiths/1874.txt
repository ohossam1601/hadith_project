١٨٧٤ - (م) عن سلمـة بن الأكـوع، قال: قـَدِ مـْنا الـحـديـبيـة مع رسول الله ﷺ ونحـنُ أر بع عشـرة مئة، وعـليـها خمسـون شاة لا تـُرْويـها، فـقـعـد رسول الله ﷺ على جـَبـا الـرَّكـِيـَّة، فـإما دعـا، وإما بـصـق فيـها، فـَجـاشـَتْ فـسـقـَيـْنا واسـْتـقـَيـْنا، ثم إن رسول الله ﷺ دعـانـا للبيـعـة في أصـل الشـجـرة، فـبـايعـتـُه في أول الناس، ثم بـايـع وبـايـع، حتى إذا كان في وسـط من النـاس قال: بـايـع یا سـلمـةُ، قلتُ: قد بـايعـتـُك يا رسول الله في أول الناس، قال: وأيضاً، قال: وقد رآني رسول الله ﷺ أَعـْـزَل - يعني: ليس معه سـلاح - فـأعـطـاني رسول الله ﷺ حـَجـَفـَة - أو دَرَقـَة - ثم بـايـع، حتى إذا كان في آخـر الناس، قال: ألا تـُبـايـعـني يا سـلمـةُ؟ قلتُ: قد بـايعـتـُك يا رسول الله في أول الناس، وفي أوسط الناس، قال: وأيضاً، فبـايعـتـُه الثـالـثـة، ثم قال لي: يا سـلمـةُ، أيـن حـَجـَفـَتُك - أو دَرَقـَتُك - التي أعطـيتُك؟ قلتُ: يا رسول الله، لقيني عـمـّـي عامـرٌ أَعـْـزَل، فـأعـطيـتـُه إياها، فضحك ﷺ وقال: إنك كالذي قال الأول: اللـهـم أبـْغـِني حـبـيـبـاً هو أحـبُّ إلـيَّ مـن نـفسـي، ثم إنَّ المشركين راسـلـونا الصـِّـلـح، حتـى مشـى بـعـضـنـا في بـعـض، واصـطـلـحـنا، وكنتُ تـَبـيـعـاً لطلحة بن عـُبيد الله، أسـقي فـرَسـه وأحـُشـُّه وأخـدمـه، وآكل من طـعـامـه، وتـركتُ أهلـي ومـالي مـهاجـراً إلى الله ورسـولـه ﷺ فلما اصـطـلـحـنا نحن وأهلُ مـكـة، واختـلط بـعـضـنـا بـبـعـض، أتيـتُ شـجـرة، فـكـسـحـْتُ شـوكـها، فـاضـطجـعـتُ في أصـلها، فـأتاني أربـعـة من المشركين من أهل مـكة، فـجـعلـوا يـقعـون في رسول الله ﷺ فـأبـْغـَـضـْتـُهـم، فـتـحـوَّلـتُ إلى شـجـرة أخـرى، وعـلـَّقـوا سـلاحـهم، واضـطجـعـوا، فـبـيـنـما هـم كذلك إذ نـادى مـُناد من أسـفل الـوادي: يا لـلـمـهاجـريـن، قـُتـِلَ ابـنُ زُنـَيـْم، فـاخـتـرطـتُ سـيـفـي، ثم شـدَدتُ على أولـئك الأربـعـةِ وهم رُقـُود، فـأخـذت سـلاحـهم، فـجـعلـتـُه ضـِغـثـاً في يـدي، ثم قلتُ: والذي كـَرَّم وجـه محـمـدٍ ﷺ لا يـرفع أحـد منكم رأسـَه، إلا ضـربـتُ الذي فيه عينـاه، ثم جئـتُ بـهم أسـوقـهم إلى رسول الله ﷺ وجـاء عـمـّـي عامـر بـرجـل من الـعـَبـَلات يقال له: مـِكـْرَز، يـقـودُه إلى رسول الله ﷺ عـلى فـَرَس مـُجـَفـَّف في سـبـعـين من المشركين، فـنـظر إليهم رسول الله ﷺ فـقال: دعـوهم، يـكن لـهم بـَدْءُ الـفـجـور و ثـِـنـَـاه، فـعـفـا عـنـهم و أنـزل الله ﷺ: ﴿وَهُوَ الَّذِي كَفَّ أَيْدِيَهُمْ عَنكُمْ وَأَيْدِيَكُمْ عَنْهُمْ بِبَطْنِ مَكَّةَ مِنْ بَعْدِ أَنْ أَظْفَرَكُمْ عَلَيْهِمْ وَكَانَ اللَّهُ بِمَا تَعْمَلُونَ بَصِيرًا﴾، ثم خرجنا راجعين إلى المدينة، فنزلنا منزلاً، بيننا وبين بني لحيان جبل وهم المشركون، فاستغفر رسول الله ﷺ لمن رَقِيَ هذا الجبل الليلة، كأنه طليعة للنبي ﷺ وأصحابه، قال سلمة: فَرَقِيتُ تلك الليلة مرتين أو ثلاثاً، ثم قَدِمْنَا المدينة، فبعث رسول الله ﷺ بظهره مع رباح - غلام رسول الله ﷺ - وأنا معه، وخرجتُ معه بفرس لطلحة أَنْدِيِه مع الظهر، فلما أصبحنا إذا عبدالرحمن الفزاري قد أغار على ظهر رسول الله ﷺ فاستاقه أجمع، وقتل راعيه، فقلت: يا رباح، خذ هذا الفرس فأَبْلِغْه طلحة بن عبيدالله، وأخبر رسول الله ﷺ أنَّ المشركين قد أغاروا على سَرْحه، ثم قمتُ على أكمة، فاستقبلتُ المدينة، فناديتُ ثلاثاً: يا صَبَاحاه (وفي رواية: قال: خرجتُ من المدينة أريد الغابة، قبل أن يُـؤَذَّن بالأولى، وكانت لِقاح رسول الله ﷺ تَرْعى بذي قَرَد، فلقيني غلام لعبدالرحمن بن عوف، فقال: أُخِذَتْ لِقاح رسول الله ﷺ فقلتُ: مَن أَخَذَها؟ قال: غَطَفان، فصرختُ ثلاث صَرَخات: يا صَبَاحاه، فأسمعتُ ما بين لابتَيْ المدينة) ثم خرجت في آثار القوم أرميهم بالنَّـبْل، وأَرْتَجِز، أقول:
أنا ابـنُ الأكـْـوَعِ ... واليـوْمُ يوْمُ الـرُّضـَّـعِ
فألحقُ رجلاً منهم، فأصكُّ سهماً في رَحـْـله، حتى خَلَصَ نَصْلُ السهم إلى كتفه، قلتُ:
خـذهـا وأنا ابنُ الأكوع ... واليـوْمُ يوْمُ الـرُّضـَّـعِ
فوالله، ما زلتُ أرميهم وأعْقِرُ بهم، فإذا رجع إليَّ فارس أتيتُ شجرة، فجلستُ في أصلها، ثم رميته فعَقَرْتُه، حتى إذا تَضايق الجبل، فدخلوا في تَضايُقه عَلَوْتُ الجبل، فجعلتُ أَرْدِيهم بالحجارة، فما زلتُ كذلك أَتْبَعُهم، حتى ما خلق الله من بعير من ظهر رسول الله ﷺ إلا خَلَّفْتُه وراء ظهري، وخَلَّوا بيني وبينه، ثم أتْبَعْتُهم أرميهم، حتى أَلْقَوْا أكثر من ثلاثين بُرْدَة وثلاثين رُمْحاً، يَسْتَخِفُّون، ولا يطرحون شيئاً إلا جعلتُ عليه آراماً من الحجارة يعرفها رسول الله ﷺ وأصحابه، حتى أتوا مُتَضايقاً من ثَنِيَّة، فإذا هم قد أتاهم فلان بن بَدْر الفزاري، فجلسوا يتضحكون - يعني: يتغَدَّون - وجلستُ على رأس قَرْن، قال الفزاري: ما هذا الذي أرى؟ قالوا: لقينا من هذا البَرْح، والله ما فارقنا منذ غَلَس، يرمينا حتى انْتَزَع كلَّ شيء من أيدينا، قال: فليَقُمْ إليه نفر منكم أربعة، فَصَعِدَ إليَّ منهم أربعة في الجبل، فلما أمكنوني من الكلام، قلت: هل تعرفوني؟ قالوا: لا، ومَن أنت؟ قلتُ: أنا سلمة بن الأكوع، والذي كَرَّم وجه محمد ﷺ لا أطلب رجلاً منكم إلا أدركته، ولا يطلبني رجل منكم فيدركني، قال أحدهم: أنا أظن، فرجعوا، فما بَرِحْتُ مكاني حتى رأيتُ فوارس رسول الله ﷺ يتخلَّلون الشجر، فإذ أولهم الأخرم الأسدي، وعلى إثره أبو قتادة الأنصاري، وعلى إثره المقداد بن الأسود الكِـنـْدِيُّ، فأخذتُ بعنان الأخرم، فوَلَّوا مُدبرين، قلتُ: يا أخرم، احذَرْهم لا يقْتطـِـعُوك حتى يلحق رسول الله ﷺ وأصحابه، قال: يا سلمة إن كنتَ تؤمن بالله واليوم الآخر، وتعلم أن الجنة حقٌ، والنار حقٌ، فلا تَحُلْ بيني وبين الشهادة، فخَلَّيْتُه، فالتقى هو وعبدالرحمن، فَعَقَر بعبدالرحمن فَرَسه، وطَعَـنَه عبدالرحمن فقتله، وتَحَوَّل على فَرَسه، ولحق أبو قتادة - فارس رسول الله ﷺ - بعبدالرحمن فطعنه فقتله، فوالذي كرم وجه محمد ﷺ لَتَبـِعْتُهم أغدو على رِجْلـَيَّ، حتى ما أرى ورائي من أصحاب محمد ولا غبارهم شيئاً، حتى يعدلوا قبل غروب الشمس إلى شِعْب فيه ماء يقال له: ذو قَـرَد، ليشربوا منه وهم عِطاش، فنظروا إليَّ أعدُو وَراءهم، فحَلَّيْتُهم عنه - يعني: أَجْلَيْتُهم عنه - فما ذاقوا منه قطرة، ويخرجون فيشتدُّون في ثَنِيَّة، فأعدو فألحق رجلاً منهم، فأصُكُّـه بسهم في نُغْض كَتِفه، قلتُ: خُذها وأنا ابن الأكوع... واليوم يوم الرُّضَّع، قال: يا ثَكِلَتْه أُ مـُّه، أكـْوَعُه بُـكْرة؟ قلت: نعم يا عدُوَّ نفسه أكـْوَعُك بكرة، وأرْدَوْا فرسين على ثنية، فجئتُ بهما أسوقهما إلى رسول الله ﷺ ولحقني عامر بسطيحة فيها مـَذْقـَة من لـَبـَن وسطيحة فيها ماء، فتوضأت وشربت، ثم أتيتُ رسول الله ﷺ وهو على الماء الذي حَلَّيْتُهم عنه، فإذا رسول الله ﷺ قد أخذ تلك الإبل، وكلَّ شيء استنقذته من المشركين، وكلَّ رُمـْح وبُرْدة، وإذا بلال نحر ناقة من الإبل التي استنقذتُ من القوم، وإذا هو يَشـْـوِي لرسول الله ﷺ من كَـبـِدِها وسَـنامها، قلتُ: يا رسول الله، خَلِّني فأنْتخِبُ من القوم مئة رجل، فأتَّبع القوم، فلا يبقى منهم مُخـْبـر إلا قتلته، فضحك ﷺ حتى بَدَتْ نـواجـذه في ضوء النار، فقال: يا سلمة، أَتـُراك كنت فاعلاً؟ قلت: نعم، والذي أكرمك (وفي رواية: قلت: يا نبيَّ الله، إني قد حَمَيْتُ القوم الماء وهم عِطاش، فابعث إليهم الساعة، فقال: يا ابن الأكوع: مَلَكـتَ فأَسـْجِحْ) إنهم الآن لَـيُـقـرَوْن في أرض غَطَفان، فجاء رجل من غَطَفان، فقال: نَحَرَ لهم فلان جَزوراً، فلما كشفوا جلدها رأوا غُباراً، فقالوا: أتاكم القوم، فخرجوا هاربين، فلما أصبحنا قال رسول الله ﷺ: كان خيرَ فُرساننا اليوم أبو قتادة، وخيرَ رَجَّـالـَـتنا سلمة، ثم أعطاني ﷺ سهمين: سهم الفارس، وسهم الراجل، فجمعهما لي جميعاً، ثم أرْدَفني ﷺ وراءه على العـَضـْبـاء، راجعين إلى المدينة، فبينا نحن نسير، وكان رجل من الأنصار لا يُسبَق شـَدّاً، فجعل يقول: ألا مُـسابق إلى المدينة؟ هل من مُـسابق؟ فجعل يعيد ذلك، فلما سمعتُ كلامه، قلتُ: أما تـُكـْـرِم كريماً، ولا تـَهابُ شـريفاً؟ قال: لا، إلا أن يكون رسول الله ﷺ قلتُ: يا رسول الله، بأبي وأمي، ذَرْنـِي فـَلأُسـابـِقَ الرَّجـُل، قال: إن شئت، قلتُ: اذهب إليك، وثَـنـَيْتُ رِجلـَيَّ، فطـَفـَرْتُ فعدوتُ، فربـطتُ عليه شـَرَفاً أو شـَرَفيـن، أَسـْتـَبـقي نفسي، ثم عـَدَوْتُ في إثـره، فربطتُ عليه شـرفاً أو شرفين، ثم إني رفـعـتُ حتى أَلـْحـقـَه فأصُـكـَّه بين كتفيه، قلتُ: قد سَبـَقـْتُ والله، قال: أنا أظن، فـسـبـقـتـه إلى المدينة، فوالله، ما لبثنا إلا ثلاث ليال، حتى خرجنا إلى خيبر مع رسول الله ﷺ فجعل عمي عامر يـَرْتـَجـِزُ بالقوم:
تـالـلـه لـوْلا الله مـا اهـْتـَدَيـْنـا ... ولا تـَصـَدَّفـْنـا ولا صـَلـَّيـْنـا
ونـحـنُ عـنْ فـَضـْلـك مـا اسـتـَغـنـَيـنـا ... فـَثـَبـِّت الأقـْـدام إِن لاقـَيْـنا
وأنـْـزِلـَنْ سـَـكـيــنـَـة عـَلـَـيـْنـا
فقال رسول الله ﷺ: من هذا؟ قال: أنا عامر، قال: غـَفـَر لـك ربـُّك، وما استغفر رسول الله ﷺ لإنسان يـخـُصُّه إلا اسـتـُشـْهـد، فنادى عمرُ بنُ الخطـاب وهو على جـمل له: يا نـبيَّ الله، لولا مـَتـَّعـتـنا بعامر؟ فلما قـدِمـنا خيبر، خـرج مـَلـِكهُـم مـَرْحـَبُ يخطـرُ بسيفه، ويقول:
قـدْ عـَلـِمـَتْ خـَيـْبـَرُ أَنـِّي مـَرْحـَبُ ... شـا كـي الـسـِّـلاح بـطـَل مـُجـَرَّبُ
إِذا الـحـُـرُوبُ أَقـْـبـَلـَتْ تـَلـَهـَّبُ
وبـرز له عمي عامـر، فقال:
قـدْ عـَلـِمـَتْ خـَيـْبـَرُ أَنـِّي عـامـِرُ ... شـاكي الـسـِّـلاح بـطـَل مـُغامـِرُ
فاختلفا ضربتين، فوقع سيف مـرحـب في تـُرس عامر، وذهب عامر يـَسـْـفـُل له، فرجع بسيفه على نفسه، فقطـع أَكـْحـَلـَه، وكانت فيها نـفسـه، قال سلمة: وخرجت، فإذا نفر من أصحاب رسول الله ﷺ يقولون: بَطَلَ عَمَلُ عامر، قتل نَفْسَهُ، فأتيتُ رسول الله ﷺ وأنا أبكي، فقلتُ: يا رسول الله، بَطَلَ عَمَلُ عامر، قال ﷺ: من قال ذلك؟ قلتُ: ناس من أصحابك، قال: كذَبَ من قال ذلك، بل له أجره مرتين، ثم أرسلني إلى عليٍّ - وهو أرْمَدُ - فقال: لأُعْطِيَنَّ الرَّايَةَ رجلاً يُحِبُّ الله ورسولَه، ويُحِبُّه الله ورسولُه، فأتيتُ عليّاً، فجئتُ به أقودُه - وهو أرْمَدُ - حتى أتيتُ رسول الله ﷺ فَبَصَقَ في عَيْنَيْه، فَبَرَأ، وخرج مَرْحَب، فقال:
قَدْ عَلِمَتْ خَيْبَرُ أَنِّي مَرْحَبُ ... شاكي السلاح بطل مجرب
إذا الحروب أقبلت تلهب
فقال عليٌّ:
أنا الذي سَمَّتْني أُمِّي حَيْدَرَه ... كَلَيْثِ غاباتٍ كريهِ المَنْظَرَه
أُوفِيهِمُ بِالصَّاعِ كَيْلَ السَّنْدَرَه
فضرب رأسَ مَرْحَب، فقتله، ثم كان الفتحُ على يَدَيْه.
(قال ابن الأثير: قال الحـُميديُّ: في هذا الحديث من ذكر الإغارة على السـرْح، وقصة عامر وارتجازه، وقوله ﷺ: «لأعطين الراية»، ما قد اتفق البخاري معه على معناه، ولكن فيه من الزيادة والشرح ما يوجب كونه من أفراد مسلم، فأفردناه).
(وفي رواية للبخاري ومسلم) قال: خرجنا مع رسول الله ﷺ إلى خيبر، فسرنا ليلاً، فقال رجل من القوم لعامر بن الأكوع: ألا تُسمعنا من هـُنـيْـهاتك؟ وكان عامر رجلاً شاعراً، فنزل يحدو بالقوم، يقول:
الـلـهـم لـولا أنـتَ مـا اهـْتـَدَيـْنـا ... ولا تـَصـَدَّفـْنـا ولا صـَلـَّيـْنـا
فاغفر فداءً لك ما اتقـينا ... وثبـت الأقـدام إن لاقـينا
وألقِيَن سكينة علينا ... إنـا إذا صيـح بنا أبـينا
وبالصبـاح عـوَّلُوا عـليـنا
فقال رسول الله ﷺ: مَن هذا السائق؟ فقالوا: عامر بن الأكوع، فقال: يرحمه الله، قال رجل من القوم: وَجَبَتْ يا رسول الله؛ لولا أمتعتنا به! قال: فأتينا خيبر، فـحـاصرناهم، حتى أصابتنا مـخْـمـَصـة شديدة، ثم إن الله فتحها عليهم، فلما أمسى الناس مساء اليوم الذي فُـتـِحـت عليهم أوقدوا نيراناً كثيرة، فقال رسول الله ﷺ: ما هذه النيران؟ على أي شيء توقدون؟ قالوا: على لحم، قال: على أي لحم؟ قالوا: لحم الحُـمـر الإنسية، فقال النبي ﷺ: أهريقـوها واكـسـروها، فقال رجل: يا رسول الله، أو نـُهـريقُـها ونغـسـِلُـها؟ فقال: أو ذاك، فلما تـصـافَّ الـقومُ كان سيـفُ عامر فيه قِـصـر، فتناول به يهودياً ليضـربه، فرجع ذُبابُ سيفه، فأصاب رُكبـتـه، فمات منها، فلما قـَفـَلـُوا، قال سـلمـة: رآني رسول الله ﷺ شاحباً ساكتاً، وهو آخـذ بيدي، قال: ما لك؟ فقلت له: فداك أبي وأمي، زعموا أن عامراً حـَبـِط عـمـلـُه، قال: مَن قاله؟ قلتُ: قاله فلان وفلان، وأُسـيـد بنُ حـُضـيـر، فقال رسول الله ﷺ: كذب من قاله، إن له لَأجـْريـن، وجمع بين إصبعيه، إنه لجاهدٌ مُجاهدٌ، قلَّ عربيٌّ مشى بها مثله (وفي رواية): نشأ بها مثله (ولمسلم) قال سلمةُ: لما كان يوم خيبر قاتل أخي قتالاً شديداً مع رسول الله ﷺ فارتد عليه سـيفه فقتله، فقال أصـحاب رسول الله ﷺ في ذلك - وشـكـُّوا فيه - رجل مات في سلاحه، قال سـلمـة: فـقـفـل رسول الله ﷺ من خيبر، فقلت: يا رسول الله، ائذن لي أن أر جـز بك، فـأذن له رسول الله ﷺ فقال عمر بن الخطـاب: اعلم ما تقول، فقلتُ:
والـلـه لـولا الله مـا اهـْتـَدَيـْنـا ... ولا تـَصـَدَّفـْنـا ولا صـَلـَّيـْنـا
فقال رسول الله ﷺ: صدقتَ.
فـأنـزلـَنْ سـكـيـنـة عـلـيـنا ... وثـبـت الأقـدام إن لاقـيـنـا
والـمـشـر كـون قـد بـَغـَوْا عـلـيـنـا
فلما قضيتُ رَجزي، قال رسول الله ﷺ: مـَن قال هـذا؟ قلتُ: قاله أخي، فقال رسول الله ﷺ: يرحمه الله، فقلتُ: يا رسول الله، والله إن ناساً ليهابون الصلاة عليه، يقولون: رَجـُلٌ مات بسلاحه، فقال رسول الله ﷺ: كذبوا، مات جاهـداً مجاهداً فله أجره مرتين.
(جبا الرَّكِـيـَّة، الرَّكِـيـَّة: البئر، وجباها: التراب الذي أُخرج منها وجُـعِل حولها. راسلونا الصلح: كذا في أكثر النسخ؛ من المراسلة، وفي بعضها: راشـونا، بتشديد السين، أي: بدؤونا، وفي بعضها: واسـَوْنا. تـَبـيـعاً: خادماً. كسحتُ شوكها: كنسته. الضـغـث: الحـزمة المجموعة. العـَبـلات: بطن من قريش. فرس مجـفف: عليه تجفاف، وهو ثوب يـجـلـل به الفرس؛ ليـقيـه السلاح. بدء الفجور وثـثـناه: ابتداؤه وثانيه. أنـديـه، تـنـدية الفرس: تـضميـره وإجـراؤه. الـلقـاح من النوق: الحوامل، وقيل: ذوات الألبان، وكان تلك غـزوة ذي قـرَد، وهي الغزوة التي أغـاروا فيها على لـقـاح النبي ﷺ قبل خيبر بثلاث. أرميهم وأعقر بهم، أي: أرميهم بالنبل وأعقر خيلهم، يقال: عـقـرتُ به، أي: قتلتُ مركوبـَه، وجعلته راجلاً. أرْديهم بالحجارة، أي: أرميهم بالحجارة التي تسقطهم وتنزلهم. آراماً: أعلاماً من الحجارة واحدها إرْم كـعنب وأعناب. جلست على رأس قـَرْن، بفتح ثم سكون هو كل جبل صغير منقطع عن الجبل الكبير. يوم الرُّضَّع: الرُّضَّع جـمع راضع وهو اللئيم، أي: يوم هلاك اللئام. البَرْح، بفتح فسكون: الشر والعـذاب الشديد، يقال: بـرح بارح وبـرح مـبـرّح. نُغْض كتفه: الغضروف العريض في أعلاه. أكـوعُه بُـكْرة؟ برفع العين أي: أنت الأكوع الذي كنتَ بكرة هذا النهار؟ ولهذا قال نعم. فـأسـجـح: أحـسـن العفـو، وسـَهـْل الأمر، فقد قدرتَ وملكتَ. لا يُـسبق شـدّاً، يعني عدواً على الرجـليـن. طفـرتُ: وثـبتُ. ربطت عليه شـرفـاً أو شرفين أستبقي نفسي، أي: حبست نفسي عن الجري الشديد، والشرف ما ارتفع من الأرض. وقوله أستبقي نفسي بفتح الفـاء. أي: لئلا ينقطع نفسي من الإعياء. شاكي السلاح: تامُّ السلاح. يَـسـْفـُل له، بفتح الياء وإسكان السين وضم الفاء أي: يضربه من أسفله. حـيـدرة: اسم للأسد لغلظه والحادر: الغليظ القوي، وكانت أم علي سمته أول ولادته أسداً باسم جده لأمه أسد بن هشام وكان أبو طالب غائباً فلما قدم سماه علياً. السَّـنْدَرَة: مِكْيال واسع. هـُنـيْـهاتك وهـُنـيْـئاتك: ما يُـسـْتـغـْرَب منك ويُـسـْتـحـْسـن. قوله: فاغفر فداءً لك، قال المازري هذه اللفظة مشكلة فإنه لا يقال فدى الباري سبحانه وتعالى ولا يقال له سبحانه فديتك لأن ذلك إنما يستعمل في مكروه يتوقع حلوله بالشخص قال ولعل هذا وقع من غير قصد إلى حقيقة معناه كما يقال قاتله الله وتربت يداك فكان مراد الشاعر أني أبذل نفسي في رضاك وعلى كل حال فإن التجوز به يفتقر إلى ورود الشرع بالإذن فيه. ما اقتفينا: ما اكتسبنا. وَجَبَتْ، أي: وجبت الرحمة التي تـُـرْحـَم بها عليه، يعني: أنه يُـقتـل شهيداً. مـخـمـَصـة: مجـاعة. ذُبـاب السيف: طرفه الذي يُـضـْـرَب به. جاهـداً: مبالغاً في الأمر إلى آخر ما يُـجـد).
