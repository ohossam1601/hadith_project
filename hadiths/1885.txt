١٨٨٥ - (خ م) عن أبي مـوسـى الأشـعـري، قال: أرسـلـني أصـحـابي إلى رسول الله ﷺ أسـأله لـهم الـحـُمـْلان، إذ هم مـعه في جيـش الـعـُسـْرة، وهي غـزوة تـبـوك، فـقلتُ: يـا نـبيَّ الله، إنَّ أصـحـابي أرسـلـوني إلـيـك لـتـحـمـِلَـهم، فـقـال: والـلـه لا أحـمـلـكم عـلى شـيء، ووَافـَقـْتُـه وهو غـضـبـان، ولا أشـعـرُ، فـرَجـَعـْتُ حـزينـاً مـن مـَنـْع رسول الله ﷺ ومـن مـَخـافـة أن يـكـون رسول الله ﷺ قـد وَجـَد في نـفـسـه ع_لـيَّ، فـرَجـَعـْتُ إلى أصـحـابي، فـأخـبـرتُـهم الذي قال النبيُّ ﷺ فـلـم أَلـْبـَث إلا سـُوَيْـعـَة إذ سـم_عـتُ بـلالاً يـنـادي: أيَّ عـبدالله بن قـيـس؟ فـأجـبـتُـه، فـقـال: أَجِـب رسول الله ﷺ يـدْعـوك، فـلـما أت_يـتُ رسول الله ﷺ قال: خـُذ هـذین الـقـَريـنـَيـن، وهـذین الـقـريـنـین، وهـذین الـقـريـنـين، لـسـت_َّـةِ أأب_ْعـِرَة اب_ت_اع_ه_ن ح_ي_ن_ئ_ذ م_ن س_َع_ْد، ف_انط_ل_ِق ب_ه_ن_َّ إل_ى أص_ح_اب_ك، ف_ق_ل: إن_َّ الل_ه - أو ق_ال: إن_َّ رس_ول الل_ه - ي_ح_م_ل_ك_م ع_ل_ى ه_ؤلاء، ف_ار ك_ب_وه_ن_َّ، ق_ال أب_و م_وس_ى: ف_انط_ل_َق_ْـتُ إل_ى أص_ح_اب_ي ب_ه_ن_َّ، ف_ق_ل_تُ: إن_َّ رس_ول الل_ه ﷺ ي_ح_م_ل_ك_م ع_ل_ى ه_ؤلاء، ول_ك_ن وال_ل_ه، لا أدع_ك_م ح_ت_ى ي_نط_ل_ِق م_ع_ي ب_ع_ض_ك_م إل_ى م_ن س_م_ع م_ق_ال_ة رس_ول الل_ه ﷺ ح_ي_ن س_أل_ت_ه ل_ك_م، وم_َن_َع_ه ف_ي أول م_رة، ث_م إع_ط_اءه إي_َّاي ب_ع_د ذل_ك، لا ت_ظن_ُّوا أن_ي ح_دَّث_ت_ك_م ش_ي_ئ_اً ل_م ي_ق_ُل_ه، ف_ق_ال_وا ل_ي: وال_ل_ه إنك عـنـدنـا لـمـُصـَدَّق، ولـنـَفـْعـَلـَنَّ مـا أحـبـبـتَ، فـانطلـق أبو مـوسى بـنـفـر مـنـهم، حتى أتـوا الذيـن سـمـعـوا قـول رسول الله ﷺ وَمـَنـْعـِه إيـاهم، ثم إعـطـاءهـم بـعـدُ، فـحـدَّثـوهـم بـمـا حـدَّثـهم أبو مـوسـى سـواء.
(سـتـأتي رواية أخـرى للحديث في باب الأيـمان والـنـذور، وفيـها أنـه ﷺ مـا نـسـي يـميـنه ولكـنه يكـفر عـنـها ویتركـهـا. الـقـريـن: الجـمـل يـُقـرَن بـجمـل آخـر، فـهـما قـریـنـان).
