١٨٧٠ - (م) عن أنس، أن قريشاً صـالـحـوا النبي ﷺ وفيـهم سهيـل بن عمرو، فقال النبي ﷺ لِـعـَلِيٍّ: اكـتـُب: بسم الله الرحـمـن الرحيم، قال سهيـل: أمـَّا باسم الله، فـما ندري مـا بسـم الله الرحـمـن الرحـيم؟ ولكن اكـتـب مـا نـعـرف: باسمك اللـهم، فقال: اكـتب: من محمد رسول الله، قالوا: لو عَـلـِمـْنا أنـَّك رسولُ الله لاتـَّـبـعـناك، ولكن اكتب اسمـك واسـم أبيـك، فقال النبيُّ ﷺ: اكتـب: مـن مـحمد بن عبدالله، فاشترطـوا عـلى النبي ﷺ: أنَّ مـَن جـاء مـنـكم لم نـرُدُّه عـليـكم، ومـن جـاءكـم مـِـنـَّا رددتـُموه علـيـنا، فقالوا: يا رسول الله، أنـكتب هذا؟ قال: نعـم، إنـه من ذهـب مـِـنـَّا إليـهم فـأبـعـَدَه الله، ومـن جـاءنـا مـنهم، سـيـجـعل اللـه لـه فـرجـاً ومـخـرجـاً.
